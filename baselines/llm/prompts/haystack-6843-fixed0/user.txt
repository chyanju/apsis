from typing import List
from haystack import Pipeline, component
from haystack.components.routers import ConditionalRouter
from haystack.components.routers.conditional_router import Route

@component
class FilterRetriever:

    @component.output_types(documents=List[str])
    def run(self, trigger: str):
        return {'documents': ['preset']}

@component
class LookupReader:

    @component.output_types(answer=str)
    def run(self, query: str, documents: List[str]):
        return {'answer': query + ':' + ','.join(documents)}

@component
class SemanticReader:

    @component.output_types(answer=str)
    def run(self, query: str):
        return {'answer': 'semantic:' + query}

def build() -> Pipeline:
    routes = [Route(condition="{{ 'lookup' in query }}", output='{{ query }}', output_name='lookup_query', output_type=str), Route(condition="{{ 'lookup' not in query }}", output='{{ query }}', output_name='semantic_query', output_type=str)]
    p = Pipeline()
    p.add_component('router', ConditionalRouter(routes=routes))
    p.add_component('filter_retriever', FilterRetriever())
    p.add_component('lookup_reader', LookupReader())
    p.add_component('semantic_reader', SemanticReader())
    p.connect('router.lookup_query', 'lookup_reader.query')
    p.connect('router.lookup_query', 'filter_retriever.trigger')
    p.connect('filter_retriever.documents', 'lookup_reader.documents')
    p.connect('router.semantic_query', 'semantic_reader.query')
    return p
pipeline = build()
