# This is a Haystack agent workflow (Python definition).
# Analyze the workflow below for bugs, working only from this source.

from typing import List
from haystack import Pipeline, component

@component
class Source:

    @component.output_types(documents=List[str])
    def run(self, go: bool):
        return {'documents': ['hey']}

@component
class Writer:

    @component.output_types(written=int)
    def run(self, documents: List[str], policy: str='none'):
        return {'written': len(documents)}

def build() -> Pipeline:
    p = Pipeline()
    p.add_component('source', Source())
    p.add_component('writer', Writer())
    p.connect('source.documents', 'writer.documents')
    return p
pipeline = build()
