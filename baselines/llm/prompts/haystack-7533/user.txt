from haystack import Pipeline, component
from haystack.components.routers import ConditionalRouter

@component
class PromptBuilder:

    @component.output_types(prompt=str)
    def run(self, query: str='', replies: str=''):
        return {'prompt': query + replies}

@component
class Generator:

    @component.output_types(replies=str)
    def run(self, prompt: str):
        return {'replies': prompt}
_ROUTES = [{'condition': '{{answerable}}', 'output': '{{query}}', 'output_name': 'go_custom', 'output_type': str}, {'condition': '{{not answerable}}', 'output': '{{query}}', 'output_name': 'go_prompt', 'output_type': str}]

def build() -> Pipeline:
    p = Pipeline()
    p.add_component('router', ConditionalRouter(routes=_ROUTES))
    p.add_component('prompt_builder', PromptBuilder())
    p.add_component('generator', Generator())
    p.connect('router.go_prompt', 'prompt_builder.query')
    p.connect('prompt_builder.prompt', 'generator.prompt')
    p.connect('generator.replies', 'router.query')
    return p
pipeline = build()
