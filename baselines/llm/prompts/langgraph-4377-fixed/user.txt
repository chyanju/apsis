from typing import Literal, Optional
from langgraph.graph import StateGraph, START, END, MessagesState
from langgraph.prebuilt import ToolNode
from langgraph.types import Command
EXPERTS = ('addition_expert_agent', 'multiplication_expert_agent')

class SupervisorState(MessagesState):
    active_agent: Optional[str]

def add_numbers(a: int, b: int) -> int:
    return a + b

def multiply_numbers(a: int, b: int) -> int:
    return a * b

def supervisor_agent(state: SupervisorState) -> Command[Literal['addition_expert_agent', 'multiplication_expert_agent', END]]:
    agent_to_call = 'addition_expert_agent'
    if agent_to_call in EXPERTS:
        return Command(goto=agent_to_call, update={'messages': [], 'active_agent': agent_to_call})
    return {'messages': []}

def build_expert_subgraph(tool_fn):

    def call_model(state: MessagesState):
        if getattr(state['messages'][-1], 'tool_calls', None):
            return Command(goto='tool_node', update={'messages': []})
        return Command(update={'messages': []})
    tool_node = ToolNode([tool_fn])
    expert_graph = StateGraph(MessagesState)
    expert_graph.add_node('call_model', call_model)
    expert_graph.add_node('tool_node', tool_node)
    expert_graph.add_edge(START, 'call_model')
    expert_graph.add_edge('tool_node', 'call_model')
    return expert_graph.compile()
addition_expert_agent = build_expert_subgraph(add_numbers)
multiplication_expert_agent = build_expert_subgraph(multiply_numbers)

def route_entry(state: SupervisorState) -> Literal['supervisor_node', 'addition_expert_agent', 'multiplication_expert_agent']:
    return state.get('active_agent') or 'supervisor_node'

def build():
    supervisor_graph = StateGraph(SupervisorState)
    supervisor_graph.add_node('supervisor_node', supervisor_agent)
    supervisor_graph.add_node('addition_expert_agent', addition_expert_agent)
    supervisor_graph.add_node('multiplication_expert_agent', multiplication_expert_agent)
    supervisor_graph.add_conditional_edges(START, route_entry, ['supervisor_node', *EXPERTS])
    return supervisor_graph.compile()
graph = build()
