# This is a Haystack agent workflow (Python definition).
# Analyze the workflow below for bugs, working only from this source.

from haystack import Pipeline, component

@component
class GetNumbers:

    @component.output_types(first_number=int, second_number=int)
    def run(self, value: int):
        return {'first_number': value, 'second_number': value * 2}

@component
class AliceTrain:

    @component.output_types(result=int, model=str)
    def run(self, data: int):
        return {'result': data, 'model': 'alice-train'}

@component
class AlicePredict:

    @component.output_types(result=str)
    def run(self, data: int, model: str):
        return {'result': f'alice-predict-{model}-{data}'}

@component
class BobTrain:

    @component.output_types(result=int, model=str)
    def run(self, data: int, alice: str):
        return {'result': data, 'model': f'bob-train-{alice}'}

@component
class BobPredict:

    @component.output_types(result=str)
    def run(self, model: str=None):
        return {'result': f'bob-predict-{model}'}

def build() -> Pipeline:
    p = Pipeline()
    p.add_component('get_numbers_1', GetNumbers())
    p.add_component('get_numbers_2', GetNumbers())
    p.add_component('alice_train', AliceTrain())
    p.add_component('alice_predict', AlicePredict())
    p.add_component('bob_train', BobTrain())
    p.add_component('bob_predict', BobPredict())
    p.connect('get_numbers_1.first_number', 'alice_train.data')
    p.connect('get_numbers_1.second_number', 'bob_train.data')
    p.connect('get_numbers_2.second_number', 'alice_predict.data')
    p.connect('alice_predict.result', 'bob_train.alice')
    p.connect('alice_train.model', 'alice_predict.model')
    p.connect('bob_train.model', 'bob_predict.model')
    return p
pipeline = build()
