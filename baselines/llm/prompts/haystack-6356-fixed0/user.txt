from haystack import Pipeline, component
from haystack.components.routers import ConditionalRouter

@component
class PromptBuilder:

    @component.output_types(prompt=str)
    def run(self, schema: str, passage: str, replies: str=''):
        return {'prompt': schema + passage + replies}

@component
class Generator:

    @component.output_types(replies=str)
    def run(self, prompt: str):
        return {'replies': prompt}

@component
class Validator:

    @component.output_types(replies=str, valid=bool)
    def run(self, replies: str):
        return {'replies': replies, 'valid': True}

@component
class Done:

    @component.output_types(result=str)
    def run(self, replies: str):
        return {'result': replies}
_ROUTES = [{'condition': '{{valid}}', 'output': '{{replies}}', 'output_name': 'exit', 'output_type': str}, {'condition': '{{not valid}}', 'output': '{{replies}}', 'output_name': 'retry', 'output_type': str}]

def build() -> Pipeline:
    p = Pipeline()
    p.add_component('prompt_builder', PromptBuilder())
    p.add_component('generator', Generator())
    p.add_component('validator', Validator())
    p.add_component('router', ConditionalRouter(routes=_ROUTES))
    p.add_component('done', Done())
    p.connect('prompt_builder.prompt', 'generator.prompt')
    p.connect('generator.replies', 'validator.replies')
    p.connect('validator.replies', 'router.replies')
    p.connect('validator.valid', 'router.valid')
    p.connect('router.retry', 'prompt_builder.replies')
    p.connect('router.exit', 'done.replies')
    return p
pipeline = build()
