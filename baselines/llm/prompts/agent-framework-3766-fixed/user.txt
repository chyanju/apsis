using System.Runtime.CompilerServices;
using Microsoft.Agents.AI;
using Microsoft.Agents.AI.Workflows;
using Microsoft.Extensions.AI;

namespace AgentFramework3766;

public static class Case
{
    public static Workflow Build()
    {
        IChatClient chatClient = new CannedChatClient();

        AIAgent mainAgent = new ChatClientAgent(chatClient, new ChatClientAgentOptions
        {
            Id = "main-agent",
            Name = "MainAgent",
            ChatOptions = new() { Instructions = "You are a triage agent. Hand the conversation off to the agent that can help the user." },
        });

        ChatToSignalAdapter chatToSignal = new();
        RequestPort numberRequestPort = RequestPort.Create<NumberSignal, int>("GuessNumber");
        JudgeExecutor judgeExecutor = new(42);

        Workflow hitlWorkflow = new WorkflowBuilder(chatToSignal)
            .AddEdge(chatToSignal, numberRequestPort)
            .AddEdge(numberRequestPort, judgeExecutor)
            .AddEdge(judgeExecutor, numberRequestPort)
            .WithOutputFrom(judgeExecutor)
            .Build();

        ExecutorBinding hitlStep = hitlWorkflow.BindAsExecutor("HitlWorkflow");
        RequestPort externalRequestPort = RequestPort.Create<NumberSignal, int>("ExternalGuessNumber");

        return new WorkflowBuilder(mainAgent)
            .AddEdge(mainAgent, hitlStep)
            .AddEdge(hitlStep, externalRequestPort)
            .AddEdge(externalRequestPort, hitlStep)
            .WithOutputFrom(hitlStep)
            .Build();
    }
}

internal enum NumberSignal
{
    Init,
    Above,
    Below,
}

internal sealed class ChatToSignalAdapter()
    : ChatProtocolExecutor("ChatToSignalAdapter", new ChatProtocolExecutorOptions { AutoSendTurnToken = false })
{
    protected override ValueTask TakeTurnAsync(List<ChatMessage> messages, IWorkflowContext context, bool? emitEvents, CancellationToken cancellationToken = default)
        => context.SendMessageAsync(NumberSignal.Init, cancellationToken: cancellationToken);
}

internal sealed class JudgeExecutor() : Executor<int>("Judge")
{
    private readonly int _targetNumber;
    private int _tries;

    public JudgeExecutor(int targetNumber) : this()
    {
        this._targetNumber = targetNumber;
    }

    public override async ValueTask HandleAsync(int message, IWorkflowContext context, CancellationToken cancellationToken = default)
    {
        this._tries++;
        if (message == this._targetNumber)
        {
            await context.YieldOutputAsync($"{this._targetNumber} found in {this._tries} tries!", cancellationToken);
        }
        else if (message < this._targetNumber)
        {
            await context.SendMessageAsync(NumberSignal.Below, cancellationToken: cancellationToken);
        }
        else
        {
            await context.SendMessageAsync(NumberSignal.Above, cancellationToken: cancellationToken);
        }
    }
}

internal sealed class CannedChatClient : IChatClient
{
    public Task<ChatResponse> GetResponseAsync(IEnumerable<ChatMessage> messages, ChatOptions? options = null, CancellationToken cancellationToken = default)
        => Task.FromResult(new ChatResponse(Reply(options)));

    public async IAsyncEnumerable<ChatResponseUpdate> GetStreamingResponseAsync(
        IEnumerable<ChatMessage> messages,
        ChatOptions? options = null,
        [EnumeratorCancellation] CancellationToken cancellationToken = default)
    {
        await Task.Yield();
        ChatMessage reply = Reply(options);
        yield return new ChatResponseUpdate(reply.Role, reply.Contents) { MessageId = Guid.NewGuid().ToString("N") };
    }

    public object? GetService(Type serviceType, object? serviceKey = null) => null;

    public void Dispose()
    {
    }

    private static ChatMessage Reply(ChatOptions? options)
    {
        AITool? handoff = options?.Tools?.FirstOrDefault(t => t.Name.StartsWith("handoff_to_", StringComparison.Ordinal));
        return handoff is null
            ? new ChatMessage(ChatRole.Assistant, "Let's play the number guessing game.")
            : new ChatMessage(ChatRole.Assistant, [new FunctionCallContent(
                Guid.NewGuid().ToString("N"),
                handoff.Name,
                new Dictionary<string, object?> { ["reasonForHandoff"] = "The user wants to play the number guessing game." })]);
    }
}
